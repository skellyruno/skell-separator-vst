cmake_minimum_required(VERSION 3.22)
project(SkellOrtApiProbe VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED ONNXRUNTIME_ROOT)
    message(FATAL_ERROR "Provide ONNXRUNTIME_ROOT pointing to official 1.26.0 CPU SDK")
endif()
add_executable(SkellOrtApiProbe probe.cpp)
target_include_directories(SkellOrtApiProbe PRIVATE "${ONNXRUNTIME_ROOT}/include")
target_link_libraries(SkellOrtApiProbe PRIVATE "${ONNXRUNTIME_ROOT}/lib/onnxruntime.lib")
